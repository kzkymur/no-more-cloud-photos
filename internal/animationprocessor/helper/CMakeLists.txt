cmake_minimum_required(VERSION 3.20)
project(nmcp_animation_helper VERSION 1.0.0 LANGUAGES CXX)

include(CTest)

add_library(nmcp_animation_core STATIC src/core.cpp)
target_include_directories(nmcp_animation_core PUBLIC include)
target_compile_features(nmcp_animation_core PUBLIC cxx_std_20)
target_compile_options(nmcp_animation_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)

if(BUILD_TESTING)
  add_executable(nmcp-animation-core-tests tests/core_test.cpp)
  target_link_libraries(nmcp-animation-core-tests PRIVATE nmcp_animation_core)
  target_compile_options(nmcp-animation-core-tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  add_test(NAME nmcp-animation-core-tests COMMAND nmcp-animation-core-tests)
endif()

option(NMCP_BUILD_HELPER "Build the dynamically linked production helper" ON)
if(NMCP_BUILD_HELPER)
  set(NMCP_PINNED_PREFIX "" CACHE PATH "Absolute prefix containing the pinned codec closure")
  if(NOT IS_ABSOLUTE "${NMCP_PINNED_PREFIX}")
    message(FATAL_ERROR "NMCP_PINNED_PREFIX must be an absolute path")
  endif()
  set(ENV{PKG_CONFIG_PATH} "")
  set(ENV{PKG_CONFIG_LIBDIR} "${NMCP_PINNED_PREFIX}/lib/pkgconfig:${NMCP_PINNED_PREFIX}/lib64/pkgconfig")
  find_package(PkgConfig REQUIRED)
  find_path(GIF_INCLUDE_DIR gif_lib.h HINTS "${NMCP_PINNED_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
  find_library(GIF_LIBRARY gif HINTS "${NMCP_PINNED_PREFIX}/lib" "${NMCP_PINNED_PREFIX}/lib64" NO_DEFAULT_PATH REQUIRED)
  add_library(Giflib::Gif UNKNOWN IMPORTED)
  set_target_properties(Giflib::Gif PROPERTIES IMPORTED_LOCATION "${GIF_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${GIF_INCLUDE_DIR}")
  pkg_check_modules(WEBP REQUIRED IMPORTED_TARGET libwebp libwebpdemux libwebpmux)
  pkg_check_modules(HEIF REQUIRED IMPORTED_TARGET libheif)
  pkg_check_modules(AOM REQUIRED IMPORTED_TARGET aom)
  pkg_check_modules(LCMS2 REQUIRED IMPORTED_TARGET lcms2)

  add_executable(nmcp-animation-helper src/main.cpp src/codec_pipeline.cpp)
  target_compile_features(nmcp-animation-helper PRIVATE cxx_std_20)
  target_compile_options(nmcp-animation-helper PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  target_link_libraries(nmcp-animation-helper PRIVATE nmcp_animation_core
    Giflib::Gif PkgConfig::WEBP PkgConfig::HEIF PkgConfig::AOM PkgConfig::LCMS2)
  set_target_properties(nmcp-animation-helper PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

  if(BUILD_TESTING)
    add_executable(nmcp-animation-fixture-generator tests/fixture_generator.cpp)
    target_compile_features(nmcp-animation-fixture-generator PRIVATE cxx_std_20)
    target_compile_options(nmcp-animation-fixture-generator PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    target_link_libraries(nmcp-animation-fixture-generator PRIVATE PkgConfig::WEBP)
    set_target_properties(nmcp-animation-fixture-generator PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

    add_executable(nmcp-gif-fixture-generator tests/gif_fixture_generator.cpp)
    target_compile_features(nmcp-gif-fixture-generator PRIVATE cxx_std_20)
    target_compile_options(nmcp-gif-fixture-generator PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    target_link_libraries(nmcp-gif-fixture-generator PRIVATE Giflib::Gif)
    set_target_properties(nmcp-gif-fixture-generator PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

    add_executable(nmcp-webp-reference tests/webp_reference.cpp)
    target_compile_features(nmcp-webp-reference PRIVATE cxx_std_20)
    target_compile_options(nmcp-webp-reference PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    target_link_libraries(nmcp-webp-reference PRIVATE PkgConfig::WEBP)
    set_target_properties(nmcp-webp-reference PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  endif()
endif()
